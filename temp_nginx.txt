server {
    listen 80;
    listen [::]:80;
    server_name api.livego.store;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    merge_slashes off;
    server_name api.livego.store;

    ssl_certificate /etc/letsencrypt/live/api.livego.store/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/api.livego.store/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header Referrer-Policy "same-origin" always;

    # Uploads servidos diretamente pelo Nginx (volume persistente fora do container)
    location /uploads/ {
        root /app/backend;
        try_files $uri =404;
        add_header Cache-Control "public, no-cache, must-revalidate" always;
        add_header Access-Control-Allow-Origin * always;
        add_header Cross-Origin-Resource-Policy "cross-origin" always;
    }

    # WHIP Proxy
    location ^~ /api/rtc/v1/whip/ {
        proxy_pass http://127.0.0.1:1985/rtc/v1/whip/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_buffering off;
        proxy_read_timeout 60s;
        proxy_send_timeout 60s;
    }

    # WHEP Proxy
    location ^~ /api/rtc/v1/whep/ {
        proxy_pass http://127.0.0.1:1985/rtc/v1/whep/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_buffering off;
        proxy_read_timeout 60s;
        proxy_send_timeout 60s;
    }

    location /livekit {
        proxy_pass http://127.0.0.1:7880/;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_buffering off;
        proxy_read_timeout 86400s;
        proxy_send_timeout 86400s;
    }
    # HLS streams via SRS (proxy direto para evitar dupla barra)
    # Tratar //live/ (double-slash gerado pelo SRS no master playlist)
    location ~ ^/api/video/http/live//live/(.*)$ {
        proxy_pass http://127.0.0.1:8080/live/$1$is_args$args;
        proxy_http_version 1.1;

        proxy_set_header Host "127.0.0.1:8080";
        proxy_buffering off;
        add_header Access-Control-Allow-Origin * always;
    }
    location ~ ^/api/video/http/live/(.*)$ {
        proxy_pass http://127.0.0.1:8080/live/$1$is_args$args;
        proxy_http_version 1.1;
        
        proxy_set_header Host "127.0.0.1:8080";
        proxy_buffering off;
        add_header Access-Control-Allow-Origin * always;
    }
    # HLS sub-playlists e segmentos via SRS
    location ~ ^/live/(.*)$ {
        proxy_pass http://127.0.0.1:8080/live/$1$is_args$args;
        proxy_http_version 1.1;
        
        proxy_set_header Host "127.0.0.1:8080";
        proxy_buffering off;
        add_header Access-Control-Allow-Origin * always;
    }


    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 300s;
        proxy_send_timeout 300s;
        proxy_buffering off;
    }
}
